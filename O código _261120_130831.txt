class AracloudTunnel:
    def __init__(self):
        self.ssh_host = ARACLOUD_SSH_HOST
        self.ssh_port = 22
        self.username = SSH_USER
        self.pkey_path = SSH_PKEY_PATH
        self.client = paramiko.SSHClient()
        self.client.set_missing_host_key_policy(paramiko.AutoAddPolicy())
        self.connected = False
        self.lock = threading.Lock()

    def connect(self):
        with self.lock:
            if self.connected and self.client.get_transport() and self.client.get_transport().is_active():
                return True
            try:
                dizer(f"🔌 A conectar ao SSH Aracloud ({self.ssh_host})...")
                if not os.path.exists(self.pkey_path):
                    dizer(f"❌ ERRO: Chave privada nao encontrada em {self.pkey_path}")
                    return False
                pkey = paramiko.RSAKey.from_private_key_file(self.pkey_path)
                self.client.connect(self.ssh_host, port=self.ssh_port, username=self.username,
                                    pkey=pkey, timeout=15, allow_agent=False, look_for_keys=False)
                transport = self.client.get_transport()
                if transport:
                    transport.set_keepalive(15)
                self.connected = True
                dizer("✅ SSH Aracloud conectado com sucesso!")
                return True
            except Exception as e:
                dizer(f"❌ ERRO SSH: {str(e)[:100]}")
                self.connected = False
                return False

    def get_socket(self, dest_host, dest_port):
        if not self.connected:
            if not self.connect(): return None
        try:
            transport = self.client.get_transport()
            if not transport or not transport.is_active():
                self.connected = False
                if not self.connect(): return None
                transport = self.client.get_transport()
            
            channel = transport.open_channel("direct-tcpip", (dest_host, dest_port), ("127.0.0.1", 0))
            channel.settimeout(15)
            return channel
        except Exception as e:
            debug_log(f"❌ ERRO TUNNEL: {str(e)[:100]}")
            self.connected = False
            # Tentativa de reconexao imediata
            if self.connect():
                try:
                    transport = self.client.get_transport()
                    channel = transport.open_channel("direct-tcpip", (dest_host, dest_port), ("127.0.0.1", 0))
                    channel.settimeout(15)
                    return channel
                except: pass
            return None

def barra(pct, largura=20):
    pct = max(0.0, min(100.0, float(pct)))
    ch = int(round(pct / 100 * largura))
    return "🟩" * ch + "⬜" * (largura - ch) if EM_GITHUB_ACTIONS else "█" * ch + "░" * (largura - ch)
def sparkline(vals, largura=10):
    chars = "▁▂▃▄▅▆▇█"
    vals = list(vals)[-largura:]
    if not vals: return "·" * largura
    mn, mx = min(vals), max(vals)
    if mx == mn: mx = mn + 1
    return "".join(chars[int((v - mn) / (mx - mn) * 7)] for v in vals)

    def _calcular_hash(self, path):
        if not os.path.exists(path): return None
        h = hashlib.sha256()
        try:
            with open(path, "rb") as f:
                for chunk in iter(lambda: f.read(65536), b""): h.update(chunk)
            return h.hexdigest()
        except Exception: return None
    def _carregar_tudo(self):
        for nome, path in self.paths.items():
            if os.path.exists(path):
                try:
                    with open(path, "r", encoding="utf-8") as f: dados = json.load(f)
                    if isinstance(dados, dict):
                        if nome == "config":
                            for k, v in self.states[nome].items():
                                if k not in dados: dados[k] = v
                        self.states[nome] = dados
                        print(f"[CHECKPOINT] 📥 Carregado local: {nome}", flush=True)
                except Exception as e: print(f"[CHECKPOINT] ⚠️ AVISO ao carregar {nome}: {str(e)[:100]}", flush=True)
    def _salvar_um(self, nome):
        try:
            path = self.paths[nome]
            d = os.path.dirname(path)
            if d: os.makedirs(d, exist_ok=True)
            tmp = path + ".tmp"
            with GLOBAL_FILEWRITE_SEMAPHORE:

    def gerar_manifesto(self, generation_id):
        manifest = {"version": self.VERSION, "generation_id": generation_id, "generated_at": datetime.now(timezone.utc).isoformat(), "files": {}}
        for nome, path in self.paths.items():
            if os.path.exists(path):
                manifest["files"][nome] = {"path": path, "hash_sha256": self._calcular_hash(path), "size_bytes": os.path.getsize(path)}
        try:
            tmp = self.manifest_path + ".tmp"
            with GLOBAL_FILEWRITE_SEMAPHORE:
                with open(tmp, "w", encoding="utf-8") as f: json.dump(manifest, f, indent=2, ensure_ascii=False)
                os.replace(tmp, self.manifest_path)
            return manifest
        except Exception as e:
            debug_log(f"gerar_manifesto: {str(e)}")
            return None
    def validar_manifesto(self):
        if not os.path.exists(self.manifest_path): return False, "Sem manifesto"
        try:
            with open(self.manifest_path, "r", encoding="utf-8") as f: manifest = json.load(f)
        except Exception as e: return False, f"Manifesto corrompido: {e}"
        for nome, info in manifest.get("files", {}).items():
            path = info.get("path")
            expected_hash = info.get("hash_sha256")
            if not path or not os.path.exists(path): return False, f"Ficheiro ausente: {nome}"
            actual_hash = self._calcular_hash(path)
            if actual_hash != expected_hash: return False, f"Hash invalido: {nome}"
        return True, "OK"

def dizer(msg, save=True):
    if not GLOBAL:
        print(msg, flush=True)
        return
    ts = datetime.now().strftime("%H:%M:%S")
    line = "[" + ts + "] " + str(msg)
    try: GLOBAL["log_buffer"].append(line)
    except Exception: pass
    print(line, flush=True)
    if save and LOG_FILE:
        try:
            with GLOBAL_FILEWRITE_SEMAPHORE:
                with open(LOG_FILE, "a", encoding="utf-8") as f: f.write(line + "\n")
        except Exception: pass

                falhados += 1

async def verificar_blacklists(ip):

async def testar_tokens_desec():

    async def _testar_um(tok):
        if GLOBAL.get("emergency_stop"): return tok, None
        try:
            async with aiohttp.ClientSession() as session:
                async with session.get("https://desec.io/api/v1/domains/", headers={"Authorization": f"Token {tok}"}, timeout=aiohttp.ClientTimeout(total=20)) as r:
                    await asyncio.sleep(0.4)
                    if r.status == 200:
                        dados = await r.json()
                        doms = []
                        if isinstance(dados, list):
                            for d in dados:
                                if isinstance(d, dict) and d.get("name"):
                                    try: min_ttl = int(d.get("minimum_ttl") or DESEC_TTL)
                                    except Exception: min_ttl = DESEC_TTL
                                    doms.append((str(d["name"]), min_ttl))
                        return tok, doms
        except Exception as e: debug_log(f"token test {tok[:6]}...: {str(e)[:120]}")
        return tok, None
    validos, mapa_feito = [], 0
    try:
        tasks = [_testar_um(tok) for tok in DESEC_TOKENS_ALL]
        results = await asyncio.gather(*tasks)
        for tok, doms in results:
            if doms is None: continue
            validos.append(tok)
            with DOM_TOKEN_LOCK:
                for nome_dom, min_ttl in doms:
                    DOM_TOKEN[nome_dom] = tok
                    DOM_MIN_TTL[nome_dom] = min_ttl
                    mapa_feito += 1
    except Exception as e: debug_log(f"testar_tokens_desec: {str(e)}")
    DESEC_TOKENS_VALIDOS.clear()
    DESEC_TOKENS_VALIDOS.extend(validos)
    dizer(f"🔐 Tokens validos: {len(validos)}/{len(DESEC_TOKENS_ALL)} | dominios mapeados: {mapa_feito}")

    marcar_fase("FASE1_TOKENS", True, f"{len(validos)} tokens, {mapa_feito} dominios")

    try:
        ans = dns.resolver.resolve(prov, 'MX')
        mxs = sorted([(r.preference, str(r.exchange).rstrip('.')) for r in ans])

        return prov, None
    except Exception:
        fb = {"naver.com": ["mx1.mail.naver.com", "mx2.mail.naver.com"], "gmail.com": ["gmail-smtp-in.l.google.com"], "googlemail.com": ["gmail-smtp-in.l.google.com"], "mail.ru": ["mxs.mail.ru"], "bk.ru": ["mxs.mail.ru"], "list.ru": ["mxs.mail.ru"], "inbox.ru": ["mxs.mail.ru"]}
        return (prov, fb.get(prov, None))

def processar_provedor_emails_sync(args):
    prov, itens = args
    if GLOBAL.get("emergency_stop"): return prov, 0
    if CHECKPOINT.provedor_ja_baixado(prov):
        path = os.path.join(DIR_EMAILS, prov + ".txt")
        if os.path.exists(path):
            try:
                with open(path, "r", encoding="utf-8", errors="ignore") as f: n = sum(1 for _ in f)
                with GLOBAL["stats_lock"]:
                    GLOBAL["pq_downloaded"] += len(itens)
                    GLOBAL["prov_processed"] += 1
                    GLOBAL["emails_read"] += n
                return prov, n
            except Exception: pass
    path = os.path.join(DIR_EMAILS, prov + ".txt")
    seen, total, ok_pq, falha_pq = set(), 0, 0, 0
    try:
        with open(path, "w", encoding="utf-8") as fout:
            for rel, local in itens:
                if GLOBAL.get("emergency_stop"): break
                origem, veio_do_snapshot, pf = None, False, None
                try:
                    if local and os.path.exists(local):
                        origem, veio_do_snapshot = local, True
                    else:
                        origem = hf_hub_download(repo_id=HF_TRADER_REPO, filename=rel, repo_type="dataset", token=HF_TOKEN, etag_timeout=60)
                    pf = pq.ParquetFile(origem)
                    nomes = pf.schema_arrow.names
                    col = None
                    for c in nomes:
                        if str(c).lower() == "email":
                            col = c
                            break
                    if col is None:
                        for c in nomes:
                            try:
                                if str(pf.schema_arrow.field(c).type) == "string":
                                    col = c
                                    break
                            except Exception: continue
                    if col is None:
                        falha_pq += 1
                        continue
                    buf = []
                    for batch in pf.iter_batches(batch_size=10000, columns=[col]):
                        if GLOBAL.get("emergency_stop"): break
                        try: valores = batch.column(0).to_pylist()
                        except Exception: continue
                        for v in valores:
                            if v is None: continue
                            e = str(v).strip().lower()
                            if "@" not in e or e in seen: continue
                            if e.startswith("@") or e.endswith("@"): continue
                            seen.add(e)
                            buf.append(e)
                            total += 1
                            if total >= 100: break
                        if buf:
                            fout.write("\n".join(buf) + "\n")
                            buf = []
                        if total >= 100: break
                        ok_pq += 1
                except Exception as e:
                    falha_pq += 1
                    debug_log(f"parquet {rel} ({prov}): {str(e)[:200]}")
                finally:
                    pf = None
                    if APAGAR_PARQUETS_APOS_USO and veio_do_snapshot and origem:
                        try:
                            if os.path.exists(origem): os.remove(origem)
                        except Exception: pass
                if total >= 100: break
            fout.flush()
            try: os.fsync(fout.fileno())
            except Exception: pass
    except Exception as e: debug_log(f"processar {prov}: {str(e)[:200]}")
    try: gc.collect()
    except Exception: pass
    if total > 0: CHECKPOINT.registrar_provedor_baixado(prov, total, ok_pq)
    with GLOBAL["stats_lock"]:
        GLOBAL["pq_downloaded"] += ok_pq
        GLOBAL["pq_falhados"] += falha_pq
        GLOBAL["prov_processed"] += 1
        GLOBAL["emails_read"] += total
    return prov, total

def ip_publico_valido(ip):
    if not ip: return False
    ip_str = str(ip).strip()
    if not re.match(r'^[\d.]+$', ip_str): return False
    try: partes = [int(p) for p in ip_str.split(".")]
    except Exception: return False
    if len(partes) != 4: return False
    for p in partes:
        if p < 0 or p > 255: return False
    a, b = partes[0], partes[1]
    if a in (0, 127, 255): return False
    if a == 10: return False
    if a == 172 and 16 <= b <= 31: return False
    if a == 192 and b == 168: return False
    if a == 169 and b == 254: return False
    if a >= 224: return False
    if a == 198 and b in (18, 19): return False
    return True

async def desec_get_com_retry_async(url, tok, domain, max_retries=4):
    for tentativa in range(max_retries):
        if GLOBAL.get("emergency_stop"): return None, "STOP"
        agora = time.time()
        espera_min = DESEC_MIN_DELAY - (agora - GLOBAL.get("desec_last_request", 0))
        if espera_min > 0: await asyncio.sleep(espera_min)
        inicio_req = time.time()
        try:
            async with aiohttp.ClientSession() as session:
                async with session.get(url, headers={"Authorization": f"Token {tok}"}, timeout=aiohttp.ClientTimeout(total=60)) as r:
                    duracao_ms = (time.time() - inicio_req) * 1000
                    GLOBAL["desec_last_request"] = time.time()
                    body = await r.text()
                    if r.status == 200:
                        log_desec_request(domain=domain, url=url, method="GET", status_code=200, response_body=f"({len(body)} bytes)", headers={}, duration_ms=duracao_ms, attempt=tentativa + 1)
                        return r, None, body
                    if r.status == 429:
                        try: wait = int(r.headers.get("Retry-After", 30))
                        except Exception: wait = 30
                        wait = min(wait, DESEC_MAX_WAIT_429)
                        dizer(f"  ⏳ Rate limit deSEC em GET {domain}. A esperar {wait}s...")
                        await asyncio.sleep(wait + 1)
                        continue
                    if r.status in (500, 502, 503, 504):
                        await asyncio.sleep(5 * (tentativa + 1))
                        continue
                    log_desec_request(domain=domain, url=url, method="GET", status_code=r.status, response_body=body, headers=dict(r.headers), duration_ms=duracao_ms, attempt=tentativa + 1)
                    return None, f"{r.status}: {body[:150]}", None
        except Exception as e:
            debug_log(f"desec_get {url} tentativa {tentativa + 1}: {str(e)[:140]}")
            await asyncio.sleep(3 * (tentativa + 1))
    return None, "MAX_RETRIES", None

def _parse_next_link(headers):
    try:
        link = headers.get("Link", "") or ""
        for parte in link.split(","):
            if 'rel="next"' in parte:
                m = re.search(r'<([^>]+)>', parte)
                if m: return m.group(1)
    except Exception: pass
    return None

def _normalizar_record(tipo, s):
    try:
        s = str(s).strip()
        if tipo == "TXT":
            s = re.sub(r'"\s*"', '', s)
            s = s.strip().strip('"')
            s = s.replace("  ", " ").replace("\t", " ")
            return s.lower()
        if tipo == "MX":
            s = s.lower().rstrip(".")
            s = re.sub(r"\s+", " ", s.strip())
            return s
        return s.lower().rstrip(".")
    except Exception: return str(s)

def records_iguais(tipo, atuais_recs, desejados_recs):
    try:
        a = sorted(_normalizar_record(tipo, x) for x in (atuais_recs or []))
        d = sorted(_normalizar_record(tipo, x) for x in (desejados_recs or []))
        return a == d
    except Exception: return False

def dividir_rrsets_por_tamanho(rrsets):
    chunks, atual, tamanho = [], [], 2
    for rs in rrsets:
        try: tam = len(json.dumps(rs, ensure_ascii=False).encode("utf-8")) + 1
        except Exception: tam = 3000
        if atual and (tamanho + tam > DESEC_CHUNK_BYTES or len(atual) >= DESEC_MAX_RRSETS_PER_REQ):
            chunks.append(atual)
            atual, tamanho = [], 2
        atual.append(rs)
        tamanho += tam
    if atual: chunks.append(atual)
    return chunks

async def desec_put_com_retry_async(url, headers, payload, domain, max_retries=5):
    payload_preview = None
    try:
        if isinstance(payload, list) and len(payload) > 0:
            payload_preview = json.dumps(payload[0], indent=2)[:2000]
            payload_preview = f"(bulk com {len(payload)} rrsets; amostra) " + payload_preview
    except Exception as e: payload_preview = f"erro: {str(e)}"
    for tentativa in range(max_retries):
        if GLOBAL.get("emergency_stop"): return False, "STOP"
        inicio_req = time.time()
        agora = time.time()
        tempo_desde_ultima = agora - GLOBAL.get("desec_last_request", 0)
        if tempo_desde_ultima < DESEC_MIN_DELAY: await asyncio.sleep(DESEC_MIN_DELAY - tempo_desde_ultima)
        try:
            async with aiohttp.ClientSession() as session:
                async with session.put(url, headers=headers, json=payload, timeout=aiohttp.ClientTimeout(total=180)) as r:
                    duracao_ms = (time.time() - inicio_req) * 1000
                    GLOBAL["desec_last_request"] = time.time()
                    body = await r.text()
                    log_desec_request(domain=domain, url=url, method="PUT", status_code=r.status, response_body=body, headers=dict(r.headers), duration_ms=duracao_ms, attempt=tentativa + 1, payload_preview=payload_preview)
                    if r.status in (200, 201, 204): return True, r.status
                    if r.status == 429:
                        try: wait_hdr = int(r.headers.get("Retry-After", DESEC_BACKOFF_BASE))
                        except Exception: wait_hdr = DESEC_BACKOFF_BASE
                        if wait_hdr > DESEC_MAX_WAIT_429:
                            dizer(f"  🚫 Rate limit LONGO em {domain} (Retry-After={wait_hdr}s > {DESEC_MAX_WAIT_429}s). A saltar.")
                            anotacao("warning", f"deSEC rate limit longo em {domain}")
                            return False, "RATE_LIMIT_LONGO"
                        wait_calc = int(DESEC_BACKOFF_BASE * (DESEC_BACKOFF_MULTIPLIER ** tentativa))
                        wait = min(max(wait_hdr, wait_calc), DESEC_MAX_WAIT_429)
                        dizer(f"  ⏳ Rate limit deSEC em {domain} (tent {tentativa + 1}/{max_retries}). Espera {wait}s...")
                        await asyncio.sleep(wait + 1)
                        continue
                    if r.status in (500, 502, 503, 504):
                        await asyncio.sleep(5 * (tentativa + 1))
                        continue
                    if r.status == 400:
                        dizer(f"  ❌ Erro 400 deSEC em {domain}: {body[:250]}")
                        return False, f"400: {body[:250]}"
                    if r.status == 404: return False, f"404: {body[:150]}"
                    return False, f"{r.status}: {body[:150]}"
        except Exception as e:
            duracao_ms = (time.time() - inicio_req) * 1000
            log_desec_request(domain=domain, url=url, method="PUT", status_code=-1, response_body=str(e), headers={}, duration_ms=duracao_ms, attempt=tentativa + 1, payload_preview=payload_preview)
            if tentativa < max_retries - 1:
                await asyncio.sleep(3 * (tentativa + 1))
                continue
            return False, "MAX_RETRIES"
    return False, "MAX_RETRIES"

async def escrever_chunk_com_isolamento(url, headers, chunk, domain, profundidade=0):
    if not chunk: return 0, []
    if domain in GLOBAL.get("dominios_rate_limited", []):
        falhados = [(rs.get("subname"), rs.get("type"), "DOMAIN_RATE_LIMITED") for rs in chunk]
        return 0, falhados
    desec_throttle_dominio(domain)
    ok, status = await desec_put_com_retry_async(url, headers, chunk, domain, max_retries=4)
    if ok: return len(chunk), []
    if isinstance(status, str) and status == "RATE_LIMIT_LONGO":
        with GLOBAL["stats_lock"]:
            rl = GLOBAL.setdefault("dominios_rate_limited", [])
            if domain not in rl: rl.append(domain)
        falhados = [(rs.get("subname"), rs.get("type"), "RATE_LIMIT_LONGO") for rs in chunk]
        return 0, falhados
    if isinstance(status, str) and status.startswith("400") and len(chunk) > 1 and profundidade < 6:
        meio = len(chunk) // 2
        ok1, falhas1 = await escrever_chunk_com_isolamento(url, headers, chunk[:meio], domain, profundidade + 1)
        ok2, falhas2 = await escrever_chunk_com_isolamento(url, headers, chunk[meio:], domain, profundidade + 1)
        return ok1 + ok2, falhas1 + falhas2
    falhados = [(rs.get("subname"), rs.get("type"), str(status)[:80]) for rs in chunk]
    return 0, falhados

def _dividir_em_chunks_com_aspas(texto, tamanho_max=255):
    if not texto: return '""'
    if len(texto) <= tamanho_max: return '"' + texto + '"'
    chunks, restante = [], texto
    while restante:
        if len(restante) <= tamanho_max:
            chunks.append(restante)
            break
        split_idx = restante.rfind(' ', 0, tamanho_max)
        if split_idx == -1 or split_idx == 0: split_idx = tamanho_max
        chunks.append(restante[:split_idx])
        restante = restante[split_idx:].lstrip()
    return " ".join(['"' + c + '"' for c in chunks])

    full_spf = " ".join(spf_parts)
    return _dividir_em_chunks_com_aspas(full_spf, 255)

def mutar_url(url, email_recipient, identity_id):
    try:
        if not url or not url.startswith("http"): return url
        parsed = urlparse(url)
        seed = email_recipient + "|" + identity_id + "|" + str(int(time.time() * 1000) % 999999)
        hash_val = hashlib.sha256(seed.encode()).hexdigest()[:16]
        fragment_novo = "u=" + hash_val
        if parsed.fragment: novo_fragment = parsed.fragment + "&" + fragment_novo
        else: novo_fragment = fragment_novo
        parsed_novo = parsed._replace(fragment=novo_fragment)
        return urlunparse(parsed_novo)
    except Exception: return url

            mm = re.search(r'<body[^>]*>', h)
            if mm: h = h.replace(mm.group(0), mm.group(0) + "\n" + rid + "\n", 1)
        h = re.sub(r'>\s+<', lambda m: '>' + (' ' * random.randint(1, 3)) + '<', h)
        def substituir_href(match):
            prefixo, url_original, sufixo = match.group(1), match.group(2), match.group(3)
            url_mutada = mutar_url(url_original, recipient_email, conta["identity_id"])
            return prefixo + url_mutada + sufixo
        h = re.sub(r'(href=["\'])([^"\']+)(["\'])', substituir_href, h)
        def substituir_src(match):
            prefixo, url_original, sufixo = match.group(1), match.group(2), match.group(3)
            url_mutada = mutar_url(url_original, recipient_email, conta["identity_id"])
            return prefixo + url_mutada + sufixo
        h = re.sub(r'(src=["\'])([^"\']+)(["\'])', substituir_src, h)
        comentario_extra = "<!-- " + hashlib.md5((recipient_email + str(time.time())).encode()).hexdigest()[:12] + " -->"
        h = h.replace("</body>", comentario_extra + "\n</body>", 1)
        return h
    except Exception as e:
        debug_log(f"mutar_html_abracadabra: {str(e)}")
        return html_base

def aplicar_mutacao_abracadabra(html_base, conta, target_emails, subject=None):
    if subject is None: subject = CAMPAIGN["subject"]
    if isinstance(target_emails, list): email_ref = target_emails[0] if target_emails else ""
    else: email_ref = str(target_emails)
    h = mutar_html_abracadabra(html_base, conta, email_ref)
    msg = EmailMessage()
    msg['Subject'] = subject
    idx_conta = 0
    for i, c in enumerate(GLOBAL.get("contas", [])):
        if c.get("identity_id") == conta.get("identity_id"):
            idx_conta = i
            break
    display_name = obter_display_name(conta, idx_conta)
    msg['From'] = f"{display_name} <{conta['sender_email']}>"
    if isinstance(target_emails, list): msg['To'] = target_emails[0] if target_emails else ""
    else: msg['To'] = str(target_emails)
    msg['Message-ID'] = make_msgid(domain=conta['domain'])
    msg['Date'] = format_datetime(datetime.now(timezone.utc))
    msg.set_content(h, subtype='html')
    b = msg.as_bytes()
    try:
        sig = dkim.sign(message=b, selector=conta['selector'].encode(), domain=conta['domain'].encode(), privkey=conta['private_key_pem'].encode(), include_headers=[b"from", b"to", b"subject", b"date", b"message-id"])
        b = sig + b
    except Exception as e: smtp_log(f"DKIM falhou em {conta.get('identity_id')}: {str(e)}")
    return b

class OrcamentoIP:
    def __init__(self):
        self.lock = threading.Lock()
        self.data = CHECKPOINT.obter_orcamento()
    def pode_enviar(self, ip, provedor):
        with self.lock:
            ip_data = self.data.get(ip, {})
            prov_data = ip_data.get(provedor, {"enviados": 0, "ultimo_ts": 0})
            if prov_data.get("enviados", 0) >= MAX_EMAILS_IP_PROVEDOR: return False, "ORCAMENTO_ESGOTADO"
            agora = time.time()
            if agora - prov_data.get("ultimo_ts", 0) < MIN_INTERVALO_IP_PROVEDOR: return False, "INTERVALO"
            return True, "OK"
    def registar_envio(self, ip, provedor, n_emails=1):
        with self.lock:
            if ip not in self.data: self.data[ip] = {}
            if provedor not in self.data[ip]: self.data[ip][provedor] = {"enviados": 0, "ultimo_ts": 0}
            self.data[ip][provedor]["enviados"] += n_emails
            self.data[ip][provedor]["ultimo_ts"] = time.time()
    def esta_esgotado(self, ip, provedor):
        with self.lock:
            ip_data = self.data.get(ip, {})
            prov_data = ip_data.get(provedor, {})
            return prov_data.get("enviados", 0) >= MAX_EMAILS_IP_PROVEDOR
    def reset_orcamento(self, ip):
        with self.lock:
            if ip in self.data:
                for prov in self.data[ip]: self.data[ip][prov]["enviados"] = 0
    def salvar(self):
        with self.lock: CHECKPOINT.salvar_orcamento(self.data)

ORCAMENTO = OrcamentoIP()

class RCPTDinamico:
    def __init__(self):
        self.lock = threading.Lock()
        self.data = CHECKPOINT.obter_rcpt_aprendido()
        self.sucessos_consecutivos = {}
    def obter(self, provedor):
        with self.lock:
            if provedor in self.data: return self.data[provedor]
            return RCPT_LIMITS.get(provedor, RCPT_DEFAULT)
    def registar_sucesso(self, provedor):
        with self.lock:
            rcpt_atual = self.data.get(provedor, RCPT_LIMITS.get(provedor, RCPT_DEFAULT))
            rcpt_inicial = RCPT_LIMITS.get(provedor, RCPT_DEFAULT)
            self.sucessos_consecutivos[provedor] = self.sucessos_consecutivos.get(provedor, 0) + 1
            if self.sucessos_consecutivos[provedor] >= 3:
                novo = min(rcpt_atual + 5, rcpt_inicial * 2)
                if novo != rcpt_atual:
                    self.data[provedor] = novo
                    self.sucessos_consecutivos[provedor] = 0
    def registar_erro(self, provedor, tipo="too_many"):
        with self.lock:
            rcpt_atual = self.data.get(provedor, RCPT_LIMITS.get(provedor, RCPT_DEFAULT))
            if tipo == "too_many": novo = max(rcpt_atual // 2, 1)
            else: novo = max(rcpt_atual // 2, 1)
            self.data[provedor] = novo
            self.sucessos_consecutivos[provedor] = 0
    def salvar(self):
        with self.lock: CHECKPOINT.salvar_rcpt_aprendido(self.data)

RCPT_CTRL = RCPTDinamico()

def _get_tunnel_socket(dest_host, dest_port):
    return TUNNEL.get_socket(dest_host, dest_port)

def _recv_smtp_raw(sock, timeout=15):
    try: sock.settimeout(timeout)
    except: pass
    data = b""
    while True:
        try:
            chunk = sock.recv(8192)
            if not chunk: break
            data += chunk
            lines = data.split(b'\r\n')
            if len(lines) >= 2:
                last_line = lines[-2] if lines[-1] == b'' else lines[-1]
                if len(last_line) >= 3 and last_line[:3].isdigit() and last_line[3:4] == b' ':
                    break
        except socket.timeout: break
        except Exception: break
    text = data.decode('utf-8', errors='ignore').strip()
    return text

def _run_smtp_audit_sync(sock, conta, provedor, emails, mx_host, html_base, worker_id):
    results = {"aceites": 0, "rejeitados": 0, "erros": 0, "tentativas": len(emails)}
    def log(msg):
        smtp_log(f"[W{worker_id:03d}] {msg}")
    try:
        # CORRECAO: Incrementar contadores globais de tentativas e conexoes
        with GLOBAL["stats_lock"]:
            GLOBAL["smtp_tentativas"] += len(emails)
            GLOBAL["smtp_conexoes"] += 1
            GLOBAL["smtp_via_proxy"] += 1
        
        banner = _recv_smtp_raw(sock)
        log(f"<- BANNER: {banner}")
        if not banner.startswith("220"): return results

        sock.sendall(f"EHLO {conta['identity_id']}\r\n".encode())
        log(f"-> EHLO {conta['identity_id']}")
        _recv_smtp_raw(sock)

        sock.sendall(f"MAIL FROM:<{conta['sender_email']}>\r\n".encode())
        log(f"-> MAIL FROM:<{conta['sender_email']}>")
        resp = _recv_smtp_raw(sock)
        log(f"<- {resp}")
        if not resp.startswith("250"): return results

        for email in emails[:100]:
            sock.sendall(f"RCPT TO:<{email}>\r\n".encode())
            log(f"-> RCPT TO:<{email}>")
            resp = _recv_smtp_raw(sock)
            log(f"<- {resp}")
            if resp.startswith("250") or resp.startswith("251"):
                results["aceites"] += 1
                with GLOBAL["stats_lock"]: GLOBAL["smtp_aceites"] += 1
                registrar_aceite(email, provedor, conta['identity_id'], mx_host)
            else:
                results["rejeitados"] += 1
                with GLOBAL["stats_lock"]: GLOBAL["smtp_rejeitados"] += 1
                registrar_rejeicao(email, provedor, resp[:3], resp)
            time.sleep(DELAY_ENTRE_RCPT)

        sock.sendall(b"DATA\r\n")
        log("-> DATA")
        resp = _recv_smtp_raw(sock)
        log(f"<- {resp}")

        if resp.startswith("354"):
            payload = aplicar_mutacao_abracadabra(html_base, conta, emails[:1])
            if not payload.endswith(b"\r\n.\r\n"):
                if not payload.endswith(b"\r\n"): payload += b"\r\n"
                payload += b".\r\n"
            sock.sendall(payload)
            log(f"-> [PAYLOAD {len(payload)} bytes]")
            _recv_smtp_raw(sock, timeout=30)

        sock.sendall(b"QUIT\r\n")
        log(f"-> QUIT")
    except Exception as e:
        log(f"❌ ERRO FATAL: {str(e)}")
        results["erros"] += 1
        with GLOBAL["stats_lock"]: GLOBAL["smtp_erros"] += 1
    finally:
        if sock:
            try: sock.close()
            except: pass
    return results

async def validar_provedor_smtp(worker_id, conta, provedor, emails, mx_host, html_base, sem):
    async with sem:
        smtp_log(f"\n{'='*70}")
        smtp_log(f"[W{worker_id:03d}] INICIO: {provedor} | MX: {mx_host} via ARAcloud")

async def enviar_email_unico(dest, assunto, html, conta):
    try:
        dominio = dest.split("@")[-1]
        try:
            answers = dns.resolver.resolve(dominio, 'MX')
            mxs_sorted = sorted([(r.preference, str(r.exchange).rstrip('.')) for r in answers])
            mx_list = [mm[1] for mm in mxs_sorted]
        except Exception: return False
        if not mx_list: return False
        for mx_host in mx_list[:3]:
            sock = _get_tunnel_socket(mx_host, 25)
            if sock:
                try:
                    banner = _recv_smtp_raw(sock)
                    if banner.startswith("220"):
                        sock.sendall(f"EHLO {conta['identity_id']}\r\n".encode())
                        _recv_smtp_raw(sock)
                        sock.sendall(f"MAIL FROM:<{conta['sender_email']}>\r\n".encode())
                        if _recv_smtp_raw(sock).startswith("250"):
                            sock.sendall(f"RCPT TO:<{dest}>\r\n".encode())
                            if _recv_smtp_raw(sock).startswith("250"):
                                sock.sendall(b"DATA\r\n")
                                if _recv_smtp_raw(sock).startswith("354"):
                                    payload = aplicar_mutacao_abracadabra(html, conta, [dest])
                                    if not payload.endswith(b"\r\n.\r\n"):
                                        if not payload.endswith(b"\r\n"): payload += b"\r\n"
                                        payload += b".\r\n"
                                    sock.sendall(payload)
                                    fr = _recv_smtp_raw(sock, timeout=30)
                                    sock.sendall(b"QUIT\r\n")
                                    sock.close()
                                    if fr.startswith("250"): return True
                except Exception: pass
                finally:
                    try: sock.close()
                    except: pass
        return False
    except Exception as e:
        debug_log(f"enviar_email_unico: {traceback.format_exc()}")
        return False

def limpar_logs_antigos(preservar_checkpoints=True, preservar_saida=False):
    garantir_diretorios()
    if os.path.exists(LOG_DIR):
        shutil.rmtree(LOG_DIR, ignore_errors=True)
        os.makedirs(LOG_DIR, exist_ok=True)
    if os.path.exists(DIR_EMAILS):
        for f in glob.glob(os.path.join(DIR_EMAILS, "*.txt")):
            try: os.remove(f)
            except Exception: pass
    else: os.makedirs(DIR_EMAILS, exist_ok=True)
    os.makedirs(DIR_AUDIT, exist_ok=True)
    os.makedirs(DIR_PROXIES, exist_ok=True)
    if not preservar_saida:
        for f in [SAIDA_ACEITES, SAIDA_UNKNOWN, SAIDA_REJEICAOES]:
            try:
                if os.path.exists(f): os.remove(f)
            except Exception: pass
